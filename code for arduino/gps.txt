// --- 1. LIBRARIES ---
#include <TinyGPS++.h>      // The library for parsing GPS data
#include <SoftwareSerial.h> // For creating a "virtual" serial port

// --- 2. SETTINGS & GLOBAL VARIABLES ---

// What pins the GPS is connected to
static const int RX_PIN = 4; // Arduino's RX pin (connects to GPS TX)
static const int TX_PIN = 5; // Arduino's TX pin (connects to GPS RX)

// The baud rate for the GPS module. 9600 is default for NEO-6M
static const uint32_t GPS_BAUD = 9600;

// The TinyGPS++ object
TinyGPSPlus gps;

// The SoftwareSerial port
SoftwareSerial ss(RX_PIN, TX_PIN);

void setup() {
  // put your setup code here, to run once:
  
  // Start the Serial Monitor
  Serial.begin(9600);
  
  // Start the serial connection to the GPS module
  ss.begin(GPS_BAUD);

  Serial.println("GPS Test Sketch");
  Serial.println("Checking for GPS module...");
  Serial.println("NOTE: This can take several minutes to get a satellite fix!");
  Serial.println("Take the module outside or near a window.");
  Serial.println("---------------------------------");
}

void loop() {
  // put your main code here, to run repeatedly:

  // This function reads data from the GPS serial port
  // and feeds it to the gps object.
  while (ss.available() > 0) {
    if (gps.encode(ss.read())) {
      // A new, complete GPS sentence was received.
      // Now, display the new data.
      displayInfo();
    }
  }

  // This checks if the GPS has stopped sending data.
  // If 5 seconds pass and no data is received, it's probably a wiring error.
  if (millis() > 5000 && gps.charsProcessed() < 10) {
    Serial.println("ERROR: No GPS data received. Check wiring or baud rate.");
    while (true); // Stop the program
  }
}

// This function prints the GPS data to the Serial Monitor
void displayInfo() {
  Serial.print("Location: ");
  if (gps.location.isValid()) {
    Serial.print(gps.location.lat(), 6); // Latitude with 6 decimal places
    Serial.print(", ");
    Serial.print(gps.location.lng(), 6); // Longitude with 6 decimal places
  } else {
    Serial.print("INVALID (No satellite fix yet)");
  }

  Serial.print("  |  Satellites: ");
  if (gps.satellites.isValid()) {
    Serial.print(gps.satellites.value());
  } else {
    Serial.print("INVALID");
  }
  Serial.println();
}
